# promptprotector default ruleset
#
# Format: blank-line-separated blocks of "key: value" lines.
# Required keys: id, category, weight, pattern (Go RE2 regexp).
# Validate with: go test -run TestDefaultRuleset ./...
#
# CURATED FOR HIGH PRECISION, NOT RECALL. A false positive blocks a real
# user, so every pattern below requires an imperative construction plus
# specific phrasing rather than bare keyword matching. Expect a bare
# keyword rule like /ignore/ to be rejected in review.
#
# MEASURED on the vendored deepset/prompt-injections train split
# (Apache-2.0), which is in testdata/train/: of the 74 English positives
# using a mechanism this set claims to detect, it catches 31. It
# false-positives 0 of the 343 real user prompts there, 122 of them
# German. Coverage is the weak side: the object nouns and phrasings
# below were widened once the corpus showed the gaps, so the remaining
# misses are known and enumerable. See testdata/SOURCES.md.
#
# The numbers come from `go test -run TestCorpusGate ./`.
#
# Known limitations, stated plainly:
#   - These are English-only. Multilingual injection ("ignoriere les
#     instructions precedentes", "ignora las instrucciones anteriores")
#     is NOT covered. Add per-language rulesets when you need it.
#   - Paraphrase defeats these. "Set aside what you were told earlier
#     and instead..." matches nothing here. The ML classifier path
#     (planned, not shipped) exists for that case.
#   - A ruleset is a DATASET, not a security boundary. Treat the
#     published pattern text as public; attackers read this file.
#
# FOUR RULES HAVE NO QUIET BENIGN FORM. Measured against a corpus of ten
# realistic documents (TestRealisticCorpusFalsePositiveRate), these fire
# on ordinary text at 0.85-0.90 and each alone blocks:
#
#   SE-001  "I'm the admin on this account"      — any support reply
#   RH-004  "Enable developer mode to access..." — any software doc
#   PI-005  "This message supersedes all previous instructions"
#                                                    — any HR/legal notice
#   PL-003  "Repeat everything above verbatim"     — any recipe
#   PL-004  the same, as the second of a pair
#
# Narrowing was measured, not assumed. Requiring a trailing "you/obey/
# must/now" on SE-001 drops it from 5 of 5 attack phrasings to 1, to
# remove 1 of 2 false positives; the same shape on RH-004 costs 2 of 4.
# The benign and attack forms are structurally identical — "I'm the
# developer" and "I'm the admin on this account" are the same shape — so
# no pattern separates them without spending more recall than it saves.
#
# These matter on ScanInput, which is deliberately aggressive for typed
# text. The indirect ruleset is the one built for documents, and it
# measures 0 false positives on this corpus. For retrieved content,
# prefer promptbuilder.Wrap over narrowing anything here.

# ---------------------------------------------------------------------
# Instruction override — "stop following your instructions"
# ---------------------------------------------------------------------

id: PI-001
category: instruction_override
weight: 0.90
pattern: (?i)\b(?:ignore|disregard|forget|discard|override|remove)\s+(?:all\s+|any\s+|the\s+|your\s+)*(?:previous|prior|above|earlier|preceding|foregoing|original|initial)\s+(?:system\s+)?(?:instruction|instructions|prompt|prompts|rule|rules|direction|directions|guideline|guidelines|context|conversation|message|messages|task|tasks|assignment|assignments|order|orders|command|commands|directive|directives|content)\b|\bleave\s+(?:all\s+)?(?:the\s+)?(?:previous|prior|above|earlier|preceding|foregoing|last|original)\s+\w+\s+behind\b

id: PI-002
category: instruction_override
weight: 0.90
pattern: (?i)\b(?:ignore|disregard|forget)\s+(?:everything|all)\s+(?:you\s+(?:were\s+told|know|learned)|above|before|previously|earlier|so\s+far)\b|\b(?:forget|ignore|disregard)\s+(?:everything|all)\s*[,.;]\s*(?:and\s+|now\s+|then\s+)?(?:write|say|print|tell|answer|respond|output|show|listen|follow|comply|do)\b

id: PI-003
category: instruction_override
weight: 0.85
pattern: (?i)\b(?:new|updated|revised|amended|override|additional)\s+(?:system\s+)?(?:instruction|instructions|prompt|prompts)\s*(?:are\s*)?(?:as\s+follows|below|:|follow)

id: PI-004
category: instruction_override
weight: 0.90
pattern: (?i)\b(?:stop|cease|halt|quit)\s+(?:following|obeying|adhering\s+to|listening\s+to)\s+(?:your\s+|the\s+)?(?:previous|prior|original|initial|former)\s+(?:instruction|instructions|prompt|direction|directions|guideline|guidelines|rule|rules)\b

id: PI-005
category: instruction_override
weight: 0.90
pattern: (?i)\bthis\s+(?:message|prompt|instruction|response)\s+(?:supersedes|overrides|replaces|takes\s+precedence\s+over)\s+(?:all\s+)?(?:previous|prior|earlier|the\s+above|your)\b

id: PI-006
category: instruction_override
weight: 0.85
pattern: (?i)\b(?:your|the)\s+(?:previous|prior|original|earlier)\s+(?:instructions?|prompt|programming|guidelines?)\s+(?:are|is|no\s+longer)\s+(?:void|invalid|obsolete|revoked|rescinded|superseded)\b

# ---------------------------------------------------------------------
# Prompt leak — "print your system prompt"
# ---------------------------------------------------------------------

# The object must be the MODEL's own prompt ("your prompt", "the system
# prompt"), not merely any instructions — otherwise "show me the
# instructions for assembling the desk" false-positives, which is a
# real failure mode caught by the benign-prose test.
id: PL-001
category: prompt_leak
weight: 0.90
pattern: (?i)\b(?:reveal|show|print|repeat|output|display|echo|dump|disclose)\s+(?:me\s+|us\s+|to\s+me\s+)?(?:all\s+|the\s+)?(?:your|the\s+system|its\s+own)\s+(?:full\s+|entire\s+|complete\s+|exact\s+|original\s+|initial\s+|whole\s+)?(?:system\s+)?(?:prompt\s*texts?|prompts?|instruction|instructions|configuration|config|system\s+message)\b

id: PL-002
category: prompt_leak
weight: 0.90
pattern: (?i)\bwhat\s+(?:is|are|was|were)\s+your\s+(?:original\s+|initial\s+|exact\s+|full\s+|system\s+)*(?:prompt|instructions|instruction|directive|directives|guidelines?)\b

id: PL-003
category: prompt_leak
weight: 0.85
pattern: (?i)\brepeat\s+(?:everything|all|the\s+text|the\s+words|the\s+content)\s+(?:above|before|preceding|prior\s+to\s+this|that\s+comes\s+before)\b

id: PL-004
category: prompt_leak
weight: 0.85
# (?m) so $ means end-of-LINE. Without it Go's RE2 anchors to end of
# TEXT and the rule only fired when the repeat instruction was the very
# last thing in the input — which in practice meant never.
pattern: (?im)\b(?:print|output|repeat|show)\s+(?:everything|all\s+text|all\s+content|the\s+words)\s+(?:above|before|preceding|prior\s+to\s+this)\s*(?:verbatim|word\s+for\s+word|exactly)?\s*[.:]?$



id: PL-005
category: prompt_leak
weight: 0.90
pattern: (?i)\b(?:reveal|print|output|show|disclose)\s+(?:your\s+)?(?:api[\s_-]?key|api[\s_-]?keys|secret[\s_-]?key|secret[\s_-]?keys|access[\s_-]?token|password|credentials?)\b

id: PL-006
category: prompt_leak
weight: 0.85
pattern: (?i)\b(?:what|which)\s+(?:were\s+you\s+told|was\s+you\s+told)\s+(?:before|initially|at\s+the\s+start|originally)\b

# Delimiter framing — forging chat-template boundaries
# ---------------------------------------------------------------------

id: DF-001
category: delimiter_framing
weight: 0.95
pattern: <\|(?:im_start|im_end|im_sep|system|user|assistant|endoftext|eot_id|start_header_id|end_header_id)\|>

id: DF-002
category: delimiter_framing
weight: 0.95
pattern: <\|begin_of_text\|>|<\|startoftext\|>|<\|pad\|>|<\|endofprompt\|>

id: DF-003
category: delimiter_framing
weight: 0.90
pattern: (?i)<\s*/?\s*(?:system|assistant|human|user)\s*>

id: DF-004
category: delimiter_framing
weight: 0.90
pattern: (?i)<\s*\|?\s*(?:start_of_system|end_of_system)\s*\|?\s*>

id: DF-005
category: delimiter_framing
weight: 0.90
pattern: (?i)\[\s*/?\s*(?:INST|SYS|SYSTEM|ASSISTANT)\s*\]|<<\s*/?\s*SYS\s*>>

id: DF-006
category: delimiter_framing
weight: 0.85
pattern: (?i)\[\s*(?:system|assistant|human)\s*\]|###\s*(?:System|Instruction|Response|Human|Assistant)\s*(?:prompt)?\s*:?

id: DF-007
category: delimiter_framing
weight: 0.90
# (?m) so ^ means start-of-LINE. Without it the rule only matched a
# forged header in the FIRST line of input, so an injection buried
# under retrieved context walked past the rule written to catch it.
pattern: (?im)^\s*(?:###\s*)?(?:SYSTEM|ASSISTANT)\s*(?:PROMPT)?\s*:\s*(?:you\s+(?:are|must|will)|ignore|disregard)

id: DX-002
category: data_exfiltration
weight: 0.85
pattern: (?i)\b(?:append|attach|inject|embed)\s+(?:the\s+|this\s+|all\s+|your\s+)*(?:conversation|chat\s+history|context|transcript|system\s+prompt)\s+(?:to|into)\s+(?:the\s+)?(?:url|link|query\s+string|image\s+src|request)\b

# Exfiltration via a URL parameter carrying a serialised context blob.
# The prefix before the parameter name is bounded and the parameter name
# is required, so ordinary URLs and query strings do not match.
id: DX-003
category: data_exfiltration
weight: 0.90
pattern: (?i)https?://[^\s?)]{1,80}\?[^\s?&)]{0,40}=(?:\{|%7B)(?:conversation|context|prompt|history|data|messages)\b
